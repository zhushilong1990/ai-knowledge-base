    **8. backend/scripts/extract_and_embed.py** — minimal end-to-end Python script:
    ```python
    import sys, json, os, uuid
    import chromadb
    from chromadb.utils import embedding_functions
    import requests
    from pypdf import PdfReader
    from docx import Document

    SILICONFLOW_API = "https://api.siliconflow.cn/v1/embeddings"
    MODEL = "BAAI/bge-m3"
    CHROMA_PATH = "backend/data/chroma"

    def extract_text(file_path, ext):
        if ext == "pdf":
            reader = PdfReader(file_path)
            return "\n\n".join(p.extract_text() or "" for p in reader.pages)
        elif ext == "docx":
            doc = Document(file_path)
            return "\n\n".join(p.text for p in doc.paragraphs if p.text.strip())
        elif ext == "txt":
            with open(file_path, encoding="utf-8") as f:
                return f.read()
        return ""

    def chunk_text(text, chunk_size=500, overlap=50):
        # char-count heuristic: 1 token ~ 1.3 Chinese chars; 500 tokens ~ 665 chars
        char_size = int(chunk_size / 0.75)  # ~665 chars
        overlap_chars = int(overlap / 0.75)  # ~67 chars
        chunks = []
        start = 0
        while start < len(text):
            end = min(start + char_size, len(text))
            chunks.append(text[start:end])
            start += char_size - overlap_chars
        return chunks

    def get_embedding(texts, api_key):
        resp = requests.post(SILICONFLOW_API,
            headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
            json={"model": MODEL, "input": texts, "encoding_format": "float"})
        resp.raise_for_status()
        data = resp.json()["data"]
        return [item["embedding"] for item in sorted(data, key=lambda x: x["index"])]

    def main():
        req = json.load(sys.stdin)
        file_path = req["filePath"]
        kb_id = req["kbId"]
        user_id = req["userId"]
        chroma_path = req["chromaPath"]
        api_key = os.environ.get("SILICONFLOW_API_KEY", "")

        ext = file_path.rsplit(".", 1)[-1].lower()
        text = extract_text(file_path, ext)
        chunks = chunk_text(text)

        # Batch embeddings (BGE-M3 limit ~8192 tokens)
        embeddings = []
        batch_size = 10
        for i in range(0, len(chunks), batch_size):
            batch = chunks[i:i+batch_size]
            embeddings.extend(get_embedding(batch, api_key))

        doc_id = str(uuid.uuid4())
        collection_name = f"user_{user_id}_kb_{kb_id}"

        client = chromadb.PersistentClient(path=chroma_path)
        collection = client.get_or_create_collection(name=collection_name)
        collection.add(
            ids=[f"{doc_id}_chunk_{j}" for j in range(len(chunks))],
            embeddings=embeddings,
            documents=chunks,
            metadatas=[{"docId": doc_id, "kbId": kb_id, "chunkIndex": j} for j in range(len(chunks))]
        )

        print(json.dumps({"documentId": doc_id, "chunkCount": len(chunks), "status": "completed"}))

    if __name__ == "__main__":
        main()
    ```

    **Key implementation notes:**
    - Python script receives JSON via stdin, writes result to stdout (per RESEARCH.md ProcessBuilder pattern)
    - Use `python` command (not `python3`) for Windows compatibility
    - Chroma collection name: `user_{userId}_kb_{kbId}` per D-12
    - Chunking: 500 tokens / 50 overlap per D-14 (char-count heuristic)
    - ProcessBuilder timeout: 120 seconds to prevent indefinite blocking
    - Script exits non-zero on any error (Java catches this and throws RuntimeException)

    **DO NOT** implement text extraction in Java — Python pypdf/python-docx is used via ProcessBuilder per D-13
    **DO NOT** create a FastAPI server — the script is standalone, called by ProcessBuilder
    **DO NOT** skip temp file cleanup — always delete in finally block
