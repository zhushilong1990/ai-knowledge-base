    ```java
    /**
     * Chunk text into ~500-token pieces with ~50-token overlap.
     * Uses char-count heuristic: 1 token ~= 1.3 Chinese chars (1 token ~= 0.75 chars for Chinese).
     * 500 tokens ~= 665 chars, 50 tokens ~= 67 chars.
     */
    public List<String> chunkText(String text) {
        List<String> chunks = new ArrayList<>();
        if (text == null || text.isEmpty()) {
            return chunks;
        }
        int charSize = 665;       // ~500 tokens
        int overlapChars = 67;    // ~50 tokens
        int start = 0;
        while (start < text.length()) {
            int end = Math.min(start + charSize, text.length());
            String chunk = text.substring(start, end);
            // Snap to nearest sentence boundary to avoid cutting mid-sentence
            if (end < text.length()) {
                int lastPunct = Math.max(
                    chunk.lastIndexOf('。'),
                    Math.max(chunk.lastIndexOf('！'),
                        Math.max(chunk.lastIndexOf('？'),
                            Math.max(chunk.lastIndexOf('.'),
                                Math.max(chunk.lastIndexOf('!'),
                                    chunk.lastIndexOf('?'))))));
                if (lastPunct > start + charSize - 100) {  // within 100 chars of boundary
                    end = lastPunct + 1;
                    chunk = text.substring(start, end);
                }
            }
            chunks.add(chunk);
            start += charSize - overlapChars;
        }
        return chunks;
    }
    ```

    ```java
    private final WebClient webClient;

    public List<float[]> generateEmbeddings(List<String> texts) {
        Map<String, Object> request = Map.of(
            "model", "BAAI/bge-m3",
            "input", texts,
            "encoding_format", "float"
        );

        List<?> data = webClient.post()
            .uri("/embeddings")
            .contentType(MediaType.APPLICATION_JSON)
            .bodyValue(request)
            .retrieve()
            .bodyToMono(Map.class)
            .timeout(Duration.ofSeconds(30))
            .block()
            .get("data");

        return ((List<?>) data).stream()
            .map(item -> {
                @SuppressWarnings("unchecked")
                List<Number> embedding = (List<Number>) ((Map<?, ?>) item).get("embedding");
                float[] vec = new float[embedding.size()];
                for (int i = 0; i < vec.length; i++) vec[i] = embedding.get(i).floatValue();
                return vec;
            })
            .collect(Collectors.toList());
    }
    ```

    ```java
    public DocumentUploadResponse uploadDocument(MultipartFile file, Long kbId, Long userId) {
        // 1. Validate extension
        String originalFilename = file.getOriginalFilename();
        String ext = originalFilename != null ? originalFilename.substring(originalFilename.lastIndexOf('.') + 1).toLowerCase() : "";
        if (!allowedExtensions.contains(ext)) {
            throw new UnsupportedFileTypeException("Unsupported file type: " + ext);
        }
        if (file.getSize() > maxFileSize) {
            throw new FileTooLargeException("File exceeds maximum size of 10MB");
        }

        // 2. Save temp file
        String tempFilePath = saveTempFile(file, ext);

        try {
            // 3. Extract text via Python script
            String extractedText = extractTextViaPython(tempFilePath, ext);

            // 4. Chunk text in Java
            List<String> chunks = chunkText(extractedText);
            if (chunks.isEmpty()) {
                throw new EmptyDocumentException("No text content could be extracted from the document");
            }

            // 5. Generate embeddings via Silicon Flow WebClient
            List<float[]> embeddings = generateEmbeddings(chunks);

            // 6. Index to Chroma via Python script
            String docId = UUID.randomUUID().toString();
            indexToChroma(docId, kbId, userId, chunks, embeddings);

            // 7. Persist document record
            Document doc = Document.builder()
                .documentId(docId)
                .userId(userId)
                .knowledgeBaseId(kbId)
                .fileName(originalFilename)
                .fileType(ext)
                .fileSize(file.getSize())
                .chunkCount(chunks.size())
                .status("completed")
                .createdAt(LocalDateTime.now())
                .updatedAt(LocalDateTime.now())
                .build();
            documentRepository.insert(doc);

            return DocumentUploadResponse.builder()
                .documentId(docId)
                .fileName(originalFilename)
                .chunkCount(chunks.size())
                .knowledgeBaseId(kbId)
                .status("completed")
                .build();
        } finally {
            // 8. Clean up temp file
            deleteTempFile(tempFilePath);
        }
    }
    ```

    ```java
    private String extractTextViaPython(String filePath, String ext) {
        try {
            ProcessBuilder pb = new ProcessBuilder("python", "backend/scripts/extract_text.py", filePath, ext);
            pb.environment().put("SILICON_FLOW_API_KEY", siliconFlowApiKey);
            Process process = pb.start();
            String output = new String(process.getInputStream().readAllBytes(), StandardCharsets.UTF_8);
            int exitCode = process.waitFor();
            if (exitCode != 0) {
                String error = new String(process.getErrorStream().readAllBytes(), StandardCharsets.UTF_8);
                throw new TextExtractionException("Text extraction failed: " + error);
            }
            return output.trim();
        } catch (IOException | InterruptedException e) {
            throw new TextExtractionException("Text extraction failed: " + e.getMessage());
        }
    }
    ```

    5. **indexToChroma()** — calls Python script with embeddings:
    ```java
    private void indexToChroma(String docId, Long kbId, Long userId, List<String> chunks, List<float[]> embeddings) {
        try {
            List<Map<String, Object>> chunkData = new ArrayList<>();
            for (int i = 0; i < chunks.size(); i++) {
                Map<String, Object> m = new HashMap<>();
                m.put("id", docId + "_chunk_" + i);
                m.put("document", chunks.get(i));
                m.put("embedding", embeddings.get(i));
                m.put("metadata", Map.of("docId", docId, "kbId", kbId, "chunkIndex", i));
                chunkData.add(m);
            }
            Map<String, Object> payload = Map.of(
                "docId", docId, "kbId", kbId, "userId", userId,
                "chromaPath", "backend/data/chroma",
                "chunks", chunkData
            );
            ProcessBuilder pb = new ProcessBuilder("python", "backend/scripts/index_to_chroma.py");
            pb.environment().put("SILICON_FLOW_API_KEY", siliconFlowApiKey);
            Process process = pb.start();
            process.getOutputStream().write(new Gson().toJson(payload).getBytes(StandardCharsets.UTF_8));
            process.getOutputStream().close();
            int exitCode = process.waitFor();
            if (exitCode != 0) {
                String error = new String(process.getErrorStream().readAllBytes(), StandardCharsets.UTF_8);
                throw new ChromaIndexException("Chroma indexing failed: " + error);
            }
        } catch (IOException | InterruptedException e) {
            throw new ChromaIndexException("Chroma indexing failed: " + e.getMessage());
        }
    }
    ```

    ```python
    import sys
    from pypdf import PdfReader
    from docx import Document

    def extract_text(file_path, ext):
        if ext == "pdf":
            reader = PdfReader(file_path)
            return "\n\n".join(p.extract_text() or "" for p in reader.pages)
        elif ext == "docx":
            doc = Document(file_path)
            return "\n\n".join(p.text for p in doc.paragraphs if p.text.strip())
        elif ext == "txt":
            with open(file_path, encoding="utf-8") as f:
                return f.read()
        return ""

    if __name__ == "__main__":
        file_path, ext = sys.argv[1], sys.argv[2].lower()
        print(extract_text(file_path, ext))
    ```

    ```python
    def main():
        req = json.load(sys.stdin)
        doc_id = req["docId"]
        kb_id = req["kbId"]
        user_id = req["userId"]
        chroma_path = req["chromaPath"]
        chunks = req["chunks"]  # list of {id, document, embedding, metadata}

        collection_name = f"user_{user_id}_kb_{kb_id}"
        client = chromadb.PersistentClient(path=chroma_path)
        collection = client.get_or_create_collection(name=collection_name)

        ids = [c["id"] for c in chunks]
        embeddings = [c["embedding"] for c in chunks]
        documents = [c["document"] for c in chunks]
        metadatas = [c["metadata"] for c in chunks]

        collection.add(ids=ids, embeddings=embeddings, documents=documents, metadatas=metadatas)
        print(json.dumps({"status": "completed", "count": len(ids)}))

    if __name__ == "__main__":
        main()
    ```
