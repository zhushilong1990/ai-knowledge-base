    **Pinia Store (chat.js):** Enhance loadHistory to also load sessions if not loaded:
    ```javascript
    async function loadHistory(sessionId) {
      try {
        const history = await getChatHistory(sessionId)
        messages.value[sessionId] = history
        currentSessionId.value = sessionId
        // Update session in sessions list to reflect it was accessed
        const sessionIndex = sessions.value.findIndex(s => s.id === sessionId)
        if (sessionIndex > 0) {
          // Move accessed session to top
          const [session] = sessions.value.splice(sessionIndex, 1)
          sessions.value.unshift(session)
        }
      } catch (error) {
        ElMessage.error('Failed to load chat history')
      }
    }
    ```
