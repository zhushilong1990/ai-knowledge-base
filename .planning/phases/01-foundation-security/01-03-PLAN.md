    async logout() {
      try {
        await fetch('/api/auth/logout', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ refreshToken: this.refreshToken })
        })

        // Network error: proceed with local logout anyway per D-04
      }
      this.clearTokens()
    }
